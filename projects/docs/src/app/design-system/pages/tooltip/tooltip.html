Examples
Rich content Pass an <ng-template> via octTooltipContent to float a panel of arbitrary content instead of a single line. The rich path is skipped on touch — provide an inline fallback there. Hover for details Payroll Listing Summary Review payroll runs and export the listing summary for a pay period.
Usage 
<!-- Basic -->
<button octTooltip="Save changes">Save</button>

<!-- With position -->
<button octTooltip="Settings" octTooltipPosition="bottom">
  <oct-icon name="Settings" />
</button>

<!-- With color variant -->
<button octTooltip="Delete item" octTooltipColor="red">Delete</button>

<!-- With delay -->
<button octTooltip="Help" [octTooltipShowDelay]="500">?</button>

<!-- Disabled -->
<button octTooltip="Hidden" [octTooltipDisabled]="true">No tooltip</button>

<!-- Rich content (floats an ng-template panel; skipped on touch) -->
<button [octTooltipContent]="tpl" octTooltipPosition="right">Details</button>
<ng-template #tpl>…rich content…</ng-template>
